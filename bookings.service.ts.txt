59:   private parseDateRange(startDate: string, endDate: string) {
60:     const start = new Date(startDate);
61:     const end = new Date(endDate);
62:     if (Number.isNaN(start.getTime()) || Number.isNaN(end.getTime())) {
63:       throw new BadRequestException("Invalid rental dates.");
64:     }
65:     const startDay = new Date(Date.UTC(start.getUTCFullYear(), start.getUTCMonth(), start.getUTCDate()));
66:     const endDay = new Date(Date.UTC(end.getUTCFullYear(), end.getUTCMonth(), end.getUTCDate()));
67:     if (endDay < startDay) throw new BadRequestException("End date must be on or after start date.");
68:     const days = Math.round((endDay.getTime() - startDay.getTime()) / MS_PER_DAY) + 1;
69:     return { startDay, endDay, days };
70:   }
